#include<bits/stdc++.h>
using namespace std;
typedef long long ll;

void solve(){
    int n; cin>>n;
    vector<ll> a(n+1,0);
    for(int i=1;i<=n;i++){
        cin>>a[i];
    }

    vector<ll> f(n+1,0);
    for(int i=1;i<=n;i++){
        if(i%2==1){
            f[i]=f[i-1]+a[i];
        }else{
            f[i]=f[i-1]-a[i];
        }
    }

    map<ll,vector<int>> pos;
    for(int i=1;i<=n;i++){
        pos[a[i]].push_back(i);
    }

    ll ans=1e18;
    for(auto &it:pos){
        vector<int> &p=it.second;
        ll sum=0;
        int prev=0;
        ll sign=1;
        for(int i=0;i<p.size();i++){
            int cur=p[i];
            if(prev+1<=cur-1){
                sum+=sign*(f[cur-1]-f[prev]);
            }
            sign*=-1;
            prev=cur;
        }
        if(prev+1<=n){
            sum+=sign*(f[n]-f[prev]);
        }
        ans=min(ans,llabs(sum));
    }
    cout<<ans<<endl;
}

int main(){
    solve();
    return 0;
}
